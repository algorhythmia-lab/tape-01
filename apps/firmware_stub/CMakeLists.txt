# As a root subdirectory, build only the portable memory/format contract probe.
# As a standalone source tree, select the Pico toolchain before project().
cmake_minimum_required(VERSION 3.22)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(PICO_BOARD pico2 CACHE STRING "Reference SDK board for RP2350 bring-up")
    set(PICO_PLATFORM rp2350-arm-s CACHE STRING "RP2350 Cortex-M33 secure target")
    if(NOT DEFINED PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
        set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
    endif()
    if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
        message(FATAL_ERROR "Supply -DPICO_SDK_PATH=/absolute/path/to/pico-sdk (SDK 2.x)")
    endif()
    include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/Version.cmake")
    project(algorhythmia_firmware_stub VERSION ${ALGORHYTHMIA_VERSION} LANGUAGES C CXX ASM)
    pico_sdk_init()
    if(NOT PICO_PLATFORM STREQUAL "rp2350-arm-s")
        message(FATAL_ERROR "This scaffold validates RP2350 Cortex-M33: use rp2350-arm-s")
    endif()
    include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/ProjectOptions.cmake")
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../libs/algorhythmia_dsp"
                     "${CMAKE_CURRENT_BINARY_DIR}/algorhythmia_dsp")
    add_executable(algorhythmia_firmware_stub src/main.cpp)
    target_include_directories(algorhythmia_firmware_stub PRIVATE include)
    target_link_libraries(algorhythmia_firmware_stub PRIVATE algorhythmia::dsp pico_stdlib)
    algorhythmia_target_defaults(algorhythmia_firmware_stub)
    pico_enable_stdio_usb(algorhythmia_firmware_stub 0)
    pico_enable_stdio_uart(algorhythmia_firmware_stub 0)
    pico_add_extra_outputs(algorhythmia_firmware_stub)
else()
    add_executable(firmware_contract src/contract_check.cpp)
    target_include_directories(firmware_contract PRIVATE include)
    target_link_libraries(firmware_contract PRIVATE algorhythmia::dsp)
    algorhythmia_target_defaults(firmware_contract)
    if(BUILD_TESTING)
        add_test(NAME firmware_contract COMMAND firmware_contract)
    endif()
endif()
